% ── Page geometry ─────────────────────────────────────────────────────────────
% KOMA-Script's typearea deliberately sets a narrow classical text block,
% which leaves most of a letter page empty at 10pt. Set an explicit type
% area instead: 6.5in text width with a small inner binding offset.
\usepackage{geometry}
\geometry{letterpaper, top=1in, bottom=1in, inner=1.1in, outer=0.9in,
  headsep=16pt, footskip=28pt}
\usepackage{mathtools}
% STIX Two Math ships a broken extensible horizontal brace: the stretchy
% middle segment of \underbrace / \overbrace renders as a solid black bar.
% Pull just those glyphs from Latin Modern Math (which draws them correctly)
% while keeping STIX Two Math for all other math. Must come after the base
% \setmathfont{STIX Two Math}, which pandoc emits before header-includes.
\setmathfont{Latin Modern Math}[range={\overbrace,\underbrace,\overbracket,\underbracket}]
% The book's meta voice: one sans family for every label, tag, running head,
% caption label, and box title. Body and display titles stay in STIX Two.
\setsansfont{Avenir Next}[Scale=MatchLowercase]
\usepackage{booktabs}
\usepackage{microtype}
\usepackage{xurl}          % allow URLs to break anywhere (bibliography)
\usepackage{xcolor}

% ── Identity palette (mirrors book.css and the web edition) ───────────────────

% Restrained link colours: internal references read as ink on paper, citations
% take the accent, and external URLs a muted slate. AtBeginDocument so this
% overrides the colours pandoc's template feeds to hyperref.
\AtBeginDocument{\hypersetup{
  linkcolor=KernelInk, citecolor=KernelAccent!85!black, urlcolor=KernelSlate,
  pdfdisplaydoctitle=true}}

% ── Body typography discipline ────────────────────────────────────────────────
\clubpenalty=10000 \widowpenalty=10000 \displaywidowpenalty=6000
\raggedbottom
% Figures are referenced as "the figure below": never float them. With [H]
% every plate is a normal block in the text flow, which also removes the
% float/breakable-tcolorbox interaction that let transparent plates overlap
% the preceding paragraph (the p.213 collision).
\usepackage{float}
\floatplacement{figure}{H}
\usepackage{amsthm}
\usepackage[most]{tcolorbox}

\definecolor{kbDef}{HTML}{2F6F4F}   % definition — green

\definecolor{kbRmk}{HTML}{7A5A9E}   % remark — purple
\definecolor{kbAlgo}{HTML}{1F6F7A}  % algorithm — teal
\definecolor{kbProof}{HTML}{59616B} % proof/hint — muted

% A titled box: thin tinted frame, coloured left rule, faint tinted fill, and a
% sans-serif coloured title that sits on the box background (no separate bar).
\tcbset{kbbox/.style={
  enhanced, breakable, arc=2.5pt, boxrule=0.4pt,
  left=10pt, right=10pt, top=7pt, bottom=8pt,
  before skip=11pt, after skip=11pt,
  fonttitle=\sffamily\footnotesize\bfseries,
  titlerule=0mm, toptitle=2pt, bottomtitle=1pt,
}}
\newtcolorbox{kbdef}[1]{kbbox, colback=kbDef!5!white, colframe=kbDef!25!white, colbacktitle=kbDef!5!white, coltitle=kbDef, borderline west={2pt}{0pt}{kbDef}, title={#1}}
\newtcolorbox{kbthm}[1]{kbbox, colback=kbThm!5!white, colframe=kbThm!25!white, colbacktitle=kbThm!5!white, coltitle=kbThm, borderline west={2pt}{0pt}{kbThm}, title={#1}}
\newtcolorbox{kbex}[1]{kbbox, colback=kbEx!5!white, colframe=kbEx!25!white, colbacktitle=kbEx!5!white, coltitle=kbEx, borderline west={2pt}{0pt}{kbEx}, title={#1}}
\newtcolorbox{kbrmk}[1]{kbbox, colback=kbRmk!5!white, colframe=kbRmk!25!white, colbacktitle=kbRmk!5!white, coltitle=kbRmk, borderline west={2pt}{0pt}{kbRmk}, title={#1}}
\newtcolorbox{kbalgo}[1]{kbbox, colback=kbAlgo!5!white, colframe=kbAlgo!25!white, colbacktitle=kbAlgo!5!white, coltitle=kbAlgo, borderline west={2pt}{0pt}{kbAlgo}, title={#1}}
% Hints: a light, title-less aside with a run-in "Hint." lead.
\newtcolorbox{kbhint}{enhanced, breakable, arc=2pt, boxrule=0pt,
  colback=kbProof!6!white, colframe=kbProof!6!white,
  borderline west={2pt}{0pt}{kbProof!45!white},
  left=8pt, right=8pt, top=5pt, bottom=5pt, before skip=7pt, after skip=9pt,
  fontupper=\small, before upper={\textit{\color{kbProof}Hint.}\enspace}}

% Chapter opener: a full-width accent hairline, a sans "CHAPTER N" tag, then
% the serif title with a deep sink before the text starts.
\renewcommand\chapterlinesformat[3]{%
  \parindent0pt\raggedright
  {\color{KernelAccent}\rule{\textwidth}{1.6pt}\par}%
  \vspace{14pt}%
  \ifstr{#2}{}{}{%
    {\sffamily\footnotesize\bfseries\color{KernelAccent}%
      \MakeUppercase{Chapter}\space#2\par}%
    \vspace{10pt}}%
  {\fontsize{27}{32}\selectfont #3\par}%
  \vspace{4pt}%
}
\renewcommand*{\chapterformat}{\thechapter}

% ── Part openers ──────────────────────────────────────────────────────────────
% \kbpart{numeral}{title}{intro}: a right-hand opener page carrying the part
% numeral, title, and the part's own introduction, recorded in TOC + bookmarks.
\newcommand{\kbpart}[3]{%
  \cleardoublepage
  \thispagestyle{empty}%
  \phantomsection
  \addcontentsline{toc}{part}{#1\ ·\ #2}%
  \vspace*{0.16\textheight}
  {\parindent0pt\raggedright
    {\sffamily\footnotesize\bfseries\color{KernelAccent}\MakeUppercase{Part #1}\par}
    \vspace{10pt}
    {\color{KernelAccent}\rule{2.6cm}{2pt}\par}
    \vspace{22pt}
    {\rmfamily\bfseries\color{KernelInk}\fontsize{30}{35}\selectfont #2\par}
    \vspace{30pt}
    \begin{minipage}{0.82\textwidth}
      {\fontsize{11.5}{17}\selectfont\itshape\color{KernelMuted} #3\par}
    \end{minipage}\par}
  \cleardoublepage
}
